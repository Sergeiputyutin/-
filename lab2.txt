void shell(int* items, intcount)
{
inti, j, gap, k;
int x, a[5];
    a[0] = 9; a[1] = 5; a[2] = 3; a[3] = 2; a[4] = 1;

for (k = 0; k < 5; k++) {
        gap = a[k];
for (i = gap; i<count; ++i) {
            x = items[i];
for (j = i - gap; (j >= 0) && (x <items[j]); j = j - gap) {
items[j + gap] = items[j];
}
items[j + gap] = x;
        }
    }
}

voidqs(int* items, intleft, intright)
{
inti, j;
int x, y;
i = left; j = right;

x = items[(left + right) / 2];

do {
while ((items[i] < x) && (i<right)) i++;
while ((x <items[j]) && (j >left)) j--;

if (i<= j) {
            y = items[i];
items[i] = items[j];
items[j] = y;
i++; j--;
        }
    } while (i<= j);

if (left< j) qs(items, left, j);
if (i<right) qs(items, i, right);
}
